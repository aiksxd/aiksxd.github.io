使用说明

点击按钮改变金币数值，体验不同的动画效果。每种动画都包含粒子特效，数值增加时为绿色，减少时为红色。

// 核心动画函数

function animateValue(element, newValue, isIncrease) {

  const oldValue = parseInt(element.textContent);

  element.textContent = newValue;

  // 清除之前的动画类

  element.className = 'value';

  element.classList.add(isIncrease ? 'increase' : 'decrease');

  element.classList.add(currentAnimation + '-animation');

  // 创建粒子效果

  createParticles(element, isIncrease);

}
